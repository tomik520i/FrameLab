cmake_minimum_required(VERSION 3.20)
project(FrameLab LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FRAMELAB_OUTPUT_NAME "FrameLab" CACHE STRING "Executable basename")
set(CUDA_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/deps/nvidia/cuda_runtime/include" CACHE PATH "CUDA runtime headers")
set(IMGUI third_party/imgui)
add_executable(FrameLab WIN32 src/main.cpp src/Graphics.cpp src/Vfg.cpp src/DirectShowCapture.cpp src/DeviceEnumeration.cpp third_party/lz4/lz4.c
 ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
 ${IMGUI}/backends/imgui_impl_dx11.cpp ${IMGUI}/backends/imgui_impl_win32.cpp)
target_include_directories(FrameLab PRIVATE src ${IMGUI} ${IMGUI}/backends "${CUDA_INCLUDE_DIR}")
target_compile_definitions(FrameLab PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(FrameLab PRIVATE /W3 /EHsc /utf-8 /MP)
target_link_libraries(FrameLab PRIVATE d3d11 dxgi d3dcompiler quartz strmiids ole32 oleaut32 setupapi propsys mmdevapi winmm shell32)
set_target_properties(FrameLab PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/app")
set_target_properties(FrameLab PROPERTIES OUTPUT_NAME "${FRAMELAB_OUTPUT_NAME}")
option(FRAMELAB_BUILD_OFFLINE_BENCHMARK "Build headless recorded-pixel experiment tools" OFF)
if(FRAMELAB_BUILD_OFFLINE_BENCHMARK)
 foreach(bench IN ITEMS FrameLab-benchmark FrameLab-benchmark-legacy)
  add_executable(${bench} tools/OfflineBenchmark.cpp src/Graphics.cpp src/Vfg.cpp)
  target_include_directories(${bench} PRIVATE src "${CUDA_INCLUDE_DIR}")
  target_compile_definitions(${bench} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(${bench} PRIVATE /W3 /EHsc /utf-8)
  target_link_libraries(${bench} PRIVATE d3d11 dxgi)
 endforeach()
 target_compile_definitions(FrameLab-benchmark-legacy PRIVATE FRAMELAB_VFG_REBIND_EACH_OUTPUT)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(vfg_pair_binding_test tests/vfg_pair_binding.cpp)
 target_include_directories(vfg_pair_binding_test PRIVATE src)
 add_test(NAME vfg_pair_binding COMMAND vfg_pair_binding_test)
 add_executable(pacer_test tests/pacer.cpp)
 target_include_directories(pacer_test PRIVATE src)
 add_test(NAME presentation_pacer COMMAND pacer_test)
 add_executable(capture_modes_test tests/capture_modes.cpp)
 target_include_directories(capture_modes_test PRIVATE src)
 add_test(NAME capture_modes COMMAND capture_modes_test)
 add_executable(recording_buffer_test tests/recording_buffer.cpp third_party/lz4/lz4.c)
 target_include_directories(recording_buffer_test PRIVATE src)
 add_test(NAME lossless_recording_buffer COMMAND recording_buffer_test)
 add_executable(recording_integration_test tests/analysis_recorder.cpp third_party/lz4/lz4.c)
 target_include_directories(recording_integration_test PRIVATE src)
 target_compile_definitions(recording_integration_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(recording_integration_test PRIVATE d3d11 dxgi)
 add_executable(display_verification_test tests/display_verification.cpp)
 target_include_directories(display_verification_test PRIVATE src)
 target_compile_definitions(display_verification_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_test(NAME display_confirmations COMMAND display_verification_test)
endif()
