cmake_minimum_required(VERSION 3.20)
project(FrameLab LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FRAMELAB_OUTPUT_NAME "FrameLab" CACHE STRING "Executable basename")
set(FRAMELAB_APP_DIR "${CMAKE_SOURCE_DIR}/app" CACHE PATH "Portable application directory")
set(CUDA_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/deps/nvidia/cuda_runtime/include" CACHE PATH "CUDA runtime headers")
set(IMGUI third_party/imgui)
# Use the same OFL fonts as the website, embedded in the portable executable.
set(FRAMELAB_FONT_HEADER "${CMAKE_BINARY_DIR}/generated/FrameLabFonts.hpp")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${FRAMELAB_FONT_HEADER}" "#pragma once\nnamespace ui::fonts {\n")
foreach(font IN ITEMS "sans|font-2.ttf" "bold|font-5.ttf" "mono|font-0.ttf")
 string(REPLACE "|" ";" parts "${font}")
 list(GET parts 0 name)
 list(GET parts 1 filename)
 set(font_path "${CMAKE_SOURCE_DIR}/website/assets/fonts/${filename}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${font_path}")
 file(READ "${font_path}" bytes HEX)
 string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${bytes}")
 file(APPEND "${FRAMELAB_FONT_HEADER}" "inline constexpr unsigned char ${name}[]={${bytes}};\n")
endforeach()
file(APPEND "${FRAMELAB_FONT_HEADER}" "}\n")
add_executable(FrameLab WIN32 src/main.cpp src/Graphics.cpp src/Vfg.cpp src/Enhancer.cpp src/SubtitleTranslator.cpp src/DirectShowCapture.cpp src/DeviceEnumeration.cpp third_party/lz4/lz4.c
 ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
 ${IMGUI}/backends/imgui_impl_dx11.cpp ${IMGUI}/backends/imgui_impl_win32.cpp)
target_include_directories(FrameLab PRIVATE src ${IMGUI} ${IMGUI}/backends "${CUDA_INCLUDE_DIR}" "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(FrameLab PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(FrameLab PRIVATE /W3 /EHsc /utf-8 /MP)
target_link_libraries(FrameLab PRIVATE d3d11 dxgi d3dcompiler quartz strmiids ole32 oleaut32 setupapi propsys mmdevapi winmm shell32 mfplat mfreadwrite mfuuid windowsapp dwmapi)
set_target_properties(FrameLab PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${FRAMELAB_APP_DIR}")
set_target_properties(FrameLab PROPERTIES OUTPUT_NAME "${FRAMELAB_OUTPUT_NAME}")
add_executable(FrameLabSetup WIN32 src/Setup.cpp)
target_compile_definitions(FrameLabSetup PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(FrameLabSetup PRIVATE /W3 /EHsc /utf-8)
target_link_libraries(FrameLabSetup PRIVATE comdlg32 shell32 dwmapi)
set_target_properties(FrameLabSetup PROPERTIES OUTPUT_NAME "FrameLab-Setup" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${FRAMELAB_APP_DIR}")
add_dependencies(FrameLab FrameLabSetup)
